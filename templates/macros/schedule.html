{# Serialize the schedule fields of `s` as the inside of a JSON object (no surrounding braces). Field list is defined once; the data_schedule macro splices this output for both the current schedule and the queued upcoming sub-schedule. #} {% macro schedule_fields(s) -%} {%- set sessions_json = s.sessions | default(value=[]) | json_encode -%} {%- set access_hours_json = s.access_hours | default(value=[]) | json_encode -%} {%- set access_exceptions_json = s.access_exceptions | default(value=[]) | json_encode -%} {%- set closures_json = s.closures | default(value=[]) | json_encode -%} {%- set schedule_basis_json = s.schedule_basis | default(value="unknown") | json_encode -%} {%- set effective_start_json = s.schedule_effective | default(value="") | json_encode -%} {%- set effective_end_json = s.schedule_effective_end | default(value="") | json_encode -%} {{- '"sessions":' ~ sessions_json ~ ',"access_hours":' ~ access_hours_json ~ ',"access_exceptions":' ~ access_exceptions_json ~ ',"closures":' ~ closures_json ~ ',"schedule_basis":' ~ schedule_basis_json ~ ',"effective_start":' ~ effective_start_json ~ ',"effective_end":' ~ effective_end_json -}} {%- endmacro schedule_fields %} {% macro data_schedule(extra) -%} {%- set body = schedule_macros::schedule_fields(s=extra) -%} {%- set upcoming = extra.upcoming_schedule | default(value=false) -%} {%- if upcoming -%} {%- set upcoming_body = schedule_macros::schedule_fields(s=upcoming) -%} {%- set rendered = '{' ~ body ~ ',"upcoming_schedule":{' ~ upcoming_body ~ '}' ~ '}' -%} {{- rendered | safe -}} {%- else -%} {%- set rendered = '{' ~ body ~ '}' -%} {{- rendered | safe -}} {%- endif -%} {%- endmacro data_schedule %}
