{# Serialize the schedule fields of `s` as a JSON object (with braces). data_schedule wraps a list of these in {"schedules": [...]}. Consumed by static/js/helpers/board.mjs `resolveScheduleForDate`. ALIAS CONTRACT: callers must import this file as `schedule_macros`, e.g. `{% import "macros/schedule.html" as schedule_macros %}`. Tera's `self::` doesn't resolve to a macro's defining file when the outer macro is called from another template — it binds to the caller's scope — so data_schedule below calls the helper via the namespace alias the caller uses. Hard-coding the alias is the trade-off. #} {% macro schedule_entry(s) -%} {%- set sessions_json = s.sessions | default(value=[]) | json_encode -%} {%- set access_hours_json = s.access_hours | default(value=[]) | json_encode -%} {%- set access_exceptions_json = s.access_exceptions | default(value=[]) | json_encode -%} {%- set closures_json = s.closures | default(value=[]) | json_encode -%} {%- set schedule_basis_json = s.schedule_basis | default(value="unknown") | json_encode -%} {%- set effective_start_json = s.effective_start | default(value="") | json_encode -%} {%- set effective_end_json = s.effective_end | default(value="") | json_encode -%} {{- '{"sessions":' ~ sessions_json ~ ',"access_hours":' ~ access_hours_json ~ ',"access_exceptions":' ~ access_exceptions_json ~ ',"closures":' ~ closures_json ~ ',"schedule_basis":' ~ schedule_basis_json ~ ',"effective_start":' ~ effective_start_json ~ ',"effective_end":' ~ effective_end_json ~ '}' -}} {%- endmacro schedule_entry %} {% macro data_schedule(extra) -%} {%- set schedules = extra.schedules | default(value=[]) -%} {%- set parts = [] -%} {%- for s in schedules -%} {%- set entry = schedule_macros::schedule_entry(s=s) -%} {%- set_global parts = parts | concat(with=entry) -%} {%- endfor -%} {%- set joined = parts | join(sep=",") -%} {%- set rendered = '{"schedules":[' ~ joined ~ ']}' -%} {{- rendered | safe -}} {%- endmacro data_schedule %}
